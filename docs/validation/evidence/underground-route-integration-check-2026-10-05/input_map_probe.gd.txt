extends SceneTree
## Evidence only: read the exact platform InputMap used by the unchanged F11 regression.

func _initialize() -> void:
	var rows: Array[Dictionary] = []
	for action: StringName in InputMap.get_actions():
		for event: InputEvent in InputMap.action_get_events(action):
			var key: InputEventKey = event as InputEventKey
			if key != null:
				rows.append({"action": String(action), "text": key.as_text(),
					"keycode": int(key.keycode), "physical_keycode": int(key.physical_keycode),
					"alt": key.alt_pressed, "control": key.ctrl_pressed,
					"shift": key.shift_pressed, "meta": key.meta_pressed,
					"command_or_control_autoremap": key.command_or_control_autoremap,
					"macos_setting": ProjectSettings.has_setting("input/" + String(action) + ".macos")})
	print("INPUT_EVENT_CENSUS " + JSON.stringify({"os": OS.get_name(),
		"display": DisplayServer.get_name(), "keyboard_event_count": rows.size(), "events": rows}))
	quit(0)
