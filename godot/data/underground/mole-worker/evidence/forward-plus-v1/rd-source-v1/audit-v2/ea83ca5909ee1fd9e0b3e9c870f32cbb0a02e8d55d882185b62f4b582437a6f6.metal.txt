// Implementation of the unsigned GLSL findMSB() function
template<typename T>
inline T spvFindUMSB(T x)
{
    return select(clz(T(0)) - (clz(x) + T(1)), T(-1), x == T(0));
}

// Returns the determinant of a 2x2 matrix.
static inline __attribute__((always_inline))
float spvDet2x2(float a1, float a2, float b1, float b2)
{
    return a1 * b2 - b1 * a2;
}

// Returns the inverse of a matrix, by using the algorithm of calculating the classical
// adjoint and dividing by the determinant. The contents of the matrix are changed.
static inline __attribute__((always_inline))
float3x3 spvInverse3x3(float3x3 m)
{
    float3x3 adj;	// The adjoint matrix (inverse after dividing by determinant)

    // Create the transpose of the cofactors, as the classical adjoint of the matrix.
    adj[0][0] =  spvDet2x2(m[1][1], m[1][2], m[2][1], m[2][2]);
    adj[0][1] = -spvDet2x2(m[0][1], m[0][2], m[2][1], m[2][2]);
    adj[0][2] =  spvDet2x2(m[0][1], m[0][2], m[1][1], m[1][2]);

    adj[1][0] = -spvDet2x2(m[1][0], m[1][2], m[2][0], m[2][2]);
    adj[1][1] =  spvDet2x2(m[0][0], m[0][2], m[2][0], m[2][2]);
    adj[1][2] = -spvDet2x2(m[0][0], m[0][2], m[1][0], m[1][2]);

    adj[2][0] =  spvDet2x2(m[1][0], m[1][1], m[2][0], m[2][1]);
    adj[2][1] = -spvDet2x2(m[0][0], m[0][1], m[2][0], m[2][1]);
    adj[2][2] =  spvDet2x2(m[0][0], m[0][1], m[1][0], m[1][1]);

    // Calculate the determinant as a combination of the cofactors of the first row.
    float det = (adj[0][0] * m[0][0]) + (adj[0][1] * m[1][0]) + (adj[0][2] * m[2][0]);

    // Divide the classical adjoint matrix by the determinant.
    // If determinant is zero, matrix is not invertable, so leave it unchanged.
    return (det != 0.0f) ? (adj * (1.0f / det)) : m;
}

static inline __attribute__((always_inline))
float vec3_to_oct_lod(thread const float3& n_ddx, thread const float3& n_ddy, thread const float& pixel_size)
{
    float pixel_size_sqr = (4.0 * pixel_size) * pixel_size;
    float ddx = dot(n_ddx, n_ddx) / pixel_size_sqr;
    float ddy = dot(n_ddy, n_ddy) / pixel_size_sqr;
    float dd_sqr = fast::max(ddx, ddy);
    return 0.25 * log2(dd_sqr + 9.9999999747524270787835121154785e-07);
}

static inline __attribute__((always_inline))
float2 oct_wrap(thread const float2& v)
{
    float2 signVal;
    signVal.x = (v.x >= 0.0) ? 1.0 : (-1.0);
    signVal.y = (v.y >= 0.0) ? 1.0 : (-1.0);
    return (float2(1.0) - abs(v.yx)) * signVal;
}

static inline __attribute__((always_inline))
float2 vec3_to_oct(thread float3& n)
{
    n /= float3((abs(n.x) + abs(n.y)) + abs(n.z));
    float2 _450;
    if (n.z >= 0.0)
    {
        _450 = n.xy;
    }
    else
    {
        float2 param = n.xy;
        _450 = oct_wrap(param);
    }
    n.x = _450.x;
    n.y = _450.y;
    float3 _465 = n;
    float2 _470 = (_465.xy * 0.5) + float2(0.5);
    n.x = _470.x;
    n.y = _470.y;
    return n.xy;
}

static inline __attribute__((always_inline))
float2 vec3_to_oct_with_border(thread const float3& n, thread const float2& border_size)
{
    float3 param = n;
    float2 _482 = vec3_to_oct(param);
    float2 uv = _482;
    return (uv * border_size.y) + float2(border_size.x);
}

static inline __attribute__((always_inline))
float4 fog_process(thread const float3& vertex0, constant SceneDataBlock& scene_data_block, sampler DEFAULT_SAMPLER_LINEAR_WITH_MIPMAPS_CLAMP, texture2d_array<float> radiance_octmap, constant DirectionalLights& directional_lights)
{
    float3 fog_color = float3(scene_data_block.data.fog_light_color);
    if (scene_data_block.data.fog_aerial_perspective > 0.0)
    {
        float3 sky_fog_color = float3(0.0);
        float3 cube_view = scene_data_block.data.radiance_inverse_xform * vertex0;
        float mip_level = mix(0.14285714924335479736328125, 1.0, 1.0 - ((abs(vertex0.z) - scene_data_block.data.z_near) / (scene_data_block.data.z_far - scene_data_block.data.z_near)));
        ResType _6133;
        _6133._m0 = modf(mip_level * 7.0, _6133._m1);
        float roughness_lod = _6133._m1;
        float blend = _6133._m0;
        float3 param = dfdx(cube_view);
        float3 param_1 = dfdy(cube_view);
        float param_2 = scene_data_block.data.radiance_pixel_size;
        float cube_lod = vec3_to_oct_lod(param, param_1, param_2);
        float3 param_3 = cube_view;
        float2 param_4 = float2(scene_data_block.data.radiance_border_size, 1.0 - (scene_data_block.data.radiance_border_size * 2.0));
        float2 cube_uv = vec3_to_oct_with_border(param_3, param_4);
        float3 _6169 = float3(cube_uv, roughness_lod);
        float3 sky_sample_a = radiance_octmap.sample(DEFAULT_SAMPLER_LINEAR_WITH_MIPMAPS_CLAMP, _6169.xy, uint(rint(_6169.z)), level(cube_lod)).xyz;
        float3 _6182 = float3(cube_uv, roughness_lod + 1.0);
        float3 sky_sample_b = radiance_octmap.sample(DEFAULT_SAMPLER_LINEAR_WITH_MIPMAPS_CLAMP, _6182.xy, uint(rint(_6182.z)), level(cube_lod)).xyz;
        sky_fog_color = mix(sky_sample_a, sky_sample_b, float3(blend));
        fog_color = mix(fog_color, sky_fog_color, float3(scene_data_block.data.fog_aerial_perspective));
    }
    if (scene_data_block.data.fog_sun_scatter > 0.001000000047497451305389404296875)
    {
        float4 sun_scatter = float4(0.0);
        float sun_total = 0.0;
        float3 view = fast::normalize(vertex0);
        for (uint i = 0u; i < scene_data_block.data.directional_light_count; i++)
        {
            float3 light_color = float3(directional_lights.data[i].color) * directional_lights.data[i].energy;
            float light_amount = powr(fast::max(dot(view, float3(directional_lights.data[i].direction)), 0.0), 8.0);
            fog_color += ((light_color * light_amount) * scene_data_block.data.fog_sun_scatter);
        }
    }
    float fog_amount = 0.0;
    if (sc_use_depth_fog())
    {
        float fog_z = smoothstep(scene_data_block.data.fog_depth_begin, scene_data_block.data.fog_depth_end, length(vertex0));
        float fog_quad_amount = powr(fog_z, scene_data_block.data.fog_depth_curve) * scene_data_block.data.fog_density;
        fog_amount = fog_quad_amount;
    }

static inline __attribute__((always_inline))

static inline __attribute__((always_inline))

static inline __attribute__((always_inline))
float3 integrate_edge_hill(thread const float3& p0, thread const float3& p1)
{
    float cosTheta = dot(p0, p1);
    float x = cosTheta;
    float y = abs(x);
    float a = 5.42031002044677734375 + ((3.1282899379730224609375 + (0.090232603251934051513671875 * y)) * y);
    float b = 3.4506800174713134765625 + ((4.18813991546630859375 + y) * y);
    float theta_sintheta = a / b;
    if (x < 0.0)
    {
        theta_sintheta = (3.1415927410125732421875 * rsqrt(1.0 - (x * x))) - theta_sintheta;
    }
    return cross(p0, p1) * theta_sintheta;
}

static inline __attribute__((always_inline))
float3 fetch_ltc_lod(thread const float2& uv, thread const float4& texture_rect, thread const float& lod, thread const float& max_mipmap, texture2d<float> area_light_atlas, sampler texture_sampler)
{
    float low = fast::min(fast::max(floor(lod), 0.0), max_mipmap - 1.0);
    float high = fast::min(fast::max(floor(lod + 1.0), 1.0), max_mipmap);
    float2 sample_pos = texture_rect.xy + (fast::clamp(uv, float2(0.0), float2(1.0)) * texture_rect.zw);
    float4 sample_col_low = area_light_atlas.sample(texture_sampler, sample_pos, level(low));
    float4 sample_col_high = area_light_atlas.sample(texture_sampler, sample_pos, level(high));
    float blend = high - fast::clamp(lod, high - 1.0, high);
    float4 sample_col = mix(sample_col_high, sample_col_low, float4(blend));
    return sample_col.xyz * sample_col.w;
}
