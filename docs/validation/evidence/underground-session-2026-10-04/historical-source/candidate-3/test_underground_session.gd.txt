class ObservedGear extends Gear:
	## Preserve real equipment matching while exercising a successful observing owner method.
	var observation: Callable = Callable()
	var observations: int = 0

	func equipment_binding_matches(inventory: Inventory, directory: Directory, residents: Residents) -> bool:
		"""One real preflight result may be followed by reentry or a real collaborator rebind."""
		var matched: bool = super.equipment_binding_matches(inventory, directory, residents)
		if observation.is_valid():
			observations += 1
			var action: Callable = observation
			observation = Callable()
			action.call()
		return matched

func before_each() -> void:
	"""Generate the real starter settlement, then supply the explicit host Gear/Carry/content dependencies."""
	_host = Settlement.new()

func after_each() -> void:
	"""Release the Session before freeing its test-owned host; no authority WeakRef is cleared by the factory."""
	if _gear != null:
		_gear.observation = Callable()
	_session = null
	_exposed = null
	_content = null

	_foreign_pool = null

		_host.free()

func _configure(gear: Gear) -> StringName:
	"""Pass actual typed host inputs, with only the explicitly tested Gear argument variable."""
	return _session.configure(_host.world(), _host.world_ref(), _host.buildings(), _host.construction(),
		_host.inventory(), _host.item_definitions(), _host.residents(), _host.jobs(), _host.work(),
		_host.reservations(), _host.transforms(), gear, _carry, _host.ground_piles(), _content)


func _host_bytes() -> Array[PackedByteArray]:
	"""Independent caller-owned test snapshots include every affected actual identity/economic/pose store."""
	return [_host.directory().state_bytes(), _host.residents().state_bytes(), _host.jobs().state_bytes(),
		_host.inventory().state_bytes(), _building_bytes(), _host.construction().state_bytes(),
		_host.transforms().state_bytes(), _host.reservations().state_bytes()]


func _building_bytes() -> PackedByteArray:
	"""Buildings has no generic state_bytes API; hash every actual packed/numeric property directly."""
	var hashing: HashingContext = HashingContext.new()
	hashing.start(HashingContext.HASH_SHA256)
	for property: Dictionary in _host.buildings().get_property_list():
		var value: Variant = _host.buildings().get(property.name)
		if value is PackedByteArray or value is PackedInt32Array or value is PackedInt64Array \
				or value is int or value is bool:
			hashing.update(var_to_bytes(value))
	return hashing.finish()


func _empty_candidates() -> void:
	"""Every refused attempt drops all private allocations and original strong borrows."""
	assert_false(_session._ready, "not published")

	assert_equal(_session._routes, null, "no private Routes receiver survives refusal")

func test_actual_foundation_preserves_all_settlement_bytes_and_uses_current_catalog() -> void:

	var before: Array[PackedByteArray] = _host_bytes()

	assert_equal(_session._content, _content, "borrowed actual Content, no second image")
	assert_equal(_session.source_owner()._locations, _session.route_owner(), "permanent actual Routes receiver")
	assert_equal(_session.terrain_owner()._world, _host.world(), "same published World")
	assert_equal(_session.space_owner()._domain, _session._domain, "one retained Domain image")

	assert_equal(_session.profile_catalog().packed_memory_bytes(), Catalog.PAIRED_BANK_BYTES, "minimal exact bank configuration")
	assert_equal(_session.cold_budget().used_bytes(), 0, "no cold lease retained")
	assert_equal(_session.route_owner()._edge_capacity, 0, "no graph allocation or travel permission")
	var facts: Owner.Facts = Owner.Facts.new()
	assert_true(_session.source_owner().read_into(_host.residents().ref_of(0), facts) != &"", "no invented resident Location")
	assert_equal(_host.buildings()._spatial_authority, null, "authority composition remains root-owned")
	assert_equal(_host.construction()._modular_authority, null, "no partial one-way binding")

func test_cached_consumer_drift_refuses_current_catalog_and_restoration_retries() -> void:
	"""The actual cached consumer text is authoritative, not the still-valid file on disk."""
	var script: Script = Contacts
	var original: String = script.get_source_code()

	var code: StringName = _configure(_gear)

	assert_equal(_session.current_refusal(), &"", "restored current closure")

func test_full_world_generation_and_seed_are_not_replaceable() -> void:
	"""The same slot and type do not retain a retired World's Session identity."""
	assert_equal(_configure(_gear), &"", "actual foundation")
	var seed_value: int = _host.world()._published_seed
	_host.world()._published_seed = seed_value + 1
	assert_equal(_session.current_refusal(), &"TERRAIN_OWNER_BINDING", "changed actual World seed")
	_host.world()._published_seed = seed_value
	assert_equal(_session.current_refusal(), &"", "original seed restored")
	var original: Vector2i = _host.world_ref()
	assert_true(_host.directory().destroy(original), "retire actual full World")
	var replacement: Vector2i = _host.directory().create(Directory.KIND_WORLD)
	assert_equal(replacement.x, original.x, "same finite slot reused")
	assert_true(replacement.y != original.y, "new generation")
	assert_true(_session.current_refusal() != &"", "replacement World cannot borrow foundation")
	assert_equal(_session.profile_catalog(), null, "stale World getter closed")


func test_original_profile_bank_and_revision_survive_refused_live_reload() -> void:
	"""A successful later Profiles reload is a different source identity even on the same object."""
	assert_equal(_configure(_gear), &"", "actual foundation")
	var bytes: PackedByteArray = FileAccess.get_file_as_bytes(Catalog.WIRE_PATH)
	bytes.encode_s64(12, 2)
	var path: String = "user://ug_session_profile_revision2.ugprof"
	var file: FileAccess = FileAccess.open(path, FileAccess.WRITE)
	file.store_buffer(bytes)
	file.close()
	var hashing: HashingContext = HashingContext.new()
	hashing.start(HashingContext.HASH_SHA256)
	hashing.update(bytes)
	assert_equal(_session._profiles.load_file(path, hashing.finish().hex_encode(), 2), &"", "actual immutable loader publishes revision2")
	assert_equal(_session.current_refusal(), &"UNDERGROUND_SESSION_PROFILES", "original source bank/revision remains mandatory")
	assert_equal(_session.profile_catalog(), null, "no new source silently adopted")
	assert_equal(DirAccess.remove_absolute(path), OK, "caller-owned mutant removed")


func test_active_inventory_and_cold_or_space_operations_block_reset_without_clearing() -> void:
	"""All three existing arenas keep their original tokens and images on a refused lifecycle boundary."""
	assert_true(_host.inventory().begin().ok, "actual Inventory journal")
	assert_equal(_configure(_gear), &"UNDERGROUND_SESSION_INVENTORY", "pre-allocation journal gate")
	_host.inventory().abort()
	assert_false(_host.inventory().is_transaction_open(), "test closes its own journal")
	assert_equal(_configure(_gear), &"", "quiescent retry")
	var token: int = _session._budget.acquire(64)
	assert_true(token > 0, "actual cold token")
	assert_equal(_session.reset_refusal(), &"UNDERGROUND_SESSION_NOT_QUIESCENT", "original lease cannot be cleared")
	assert_true(_session._budget.covers(token, 64), "lease preserved")
	assert_equal(_session._budget.release(token), &"", "caller releases its token")
	var stage: Owner.Result = _session._space.begin_stage(_session._space.revision())
	assert_equal(stage.error, &"", "actual Space candidate")
	assert_equal(_session.reset_refusal(), &"UNDERGROUND_SESSION_NOT_QUIESCENT", "candidate blocks reset")
	assert_equal(_session._space._stage_token, stage.token, "original candidate survives")
	assert_true(_session._space.abort(stage.token), "caller aborts its candidate")
	assert_equal(_session.reset_refusal(), &"", "fully quiescent again")
