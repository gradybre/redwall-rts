func test_inventory_remount_reuses_identical_arena_and_rejects_capacity_change() -> void:
	"""A real new namespace can bind only after explicit retirement, without resizing endpoint columns."""
	var slots: PackedInt32Array = _f._inventory._spatial_container_slot
	assert_equal(_prepare(), &"", "original prepares")
	_f.clear_host_stores()
	assert_equal(Retirement.release_preflighted(_scope, _host, _session), &"", "original releases")
	assert_true(_f._world.generate(World.bound_request(_f._items).request).ok, "next actual World")
	var world: Vector2i = _owners.directory.create(Directory.KIND_WORLD)
	assert_false(_owners.directory.is_valid(_f._world_ref), "old World remains stale")
	var locations: Locations = _new_inventory_namespace(world)
	var adapter: Locations.InventoryLocations = Locations.InventoryLocations.new(locations)
	assert_equal(_f._inventory.bind_spatial_locations(adapter, 7).error, Inventory.REFUSE_SPATIAL_CAPACITY, "cannot shrink old arena")
	assert_equal(_f._inventory.bind_spatial_locations(adapter, 9).error, Inventory.REFUSE_SPATIAL_CAPACITY, "cannot expand old arena")
	assert_equal(_f._inventory._spatial_authority, null, "capacity refusals do not publish a link")
	assert_true(_f._inventory.bind_spatial_locations(adapter, 8).ok, "identical existing arena remounts")
	assert_equal(_f._inventory._spatial_world, world, "new exact World captured")
	slots[0] = 123
	assert_equal(_f._inventory._spatial_container_slot[0], 123, "old alias observes the identical reused packed allocation")
	slots[0] = -1
	assert_equal(_f._inventory.bind_spatial_locations(_f.adapter, 8).error, Inventory.REFUSE_SPATIAL_BINDING, "old adapter cannot replace new lifetime")
