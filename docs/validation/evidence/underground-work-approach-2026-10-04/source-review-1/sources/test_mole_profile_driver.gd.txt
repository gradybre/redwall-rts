func _source_clock_replay(actual: RouteFixture, source: Content, driver: Driver, job: Vector2i, tool: Vector2i) -> void:
	"""Render twice or not at all between ticks; both directions resolve the actual nonuniform loop."""
	var a: Vector3i = Vector3i(-512, 0, 512)
	var b: Vector3i = Vector3i(512, 0, 512)
	var first: Vector2i = actual._location(a)
	var last: Vector2i = actual._location(b)
	actual._publish_edges([actual._edge(first, last, [a, b]), actual._edge(last, first, [b, a])])
	assert_equal(actual._routes.admit_travel_actor(actual._worker, job, first, 5, 1, 2, 0, -1, tool), &"", "canonical ready actor")
	assert_equal(actual._routes.request_route(actual._worker, last, 0), &"", "actual directed approach")
	var frame: Driver.Frame = Driver.Frame.new()
	var expected: PackedInt32Array = PackedInt32Array([0, 0, 0])
	var state: PackedInt64Array = PackedInt64Array([0, 0, 0, 0])
	for tick: int in range(1, 47):
		assert_equal(actual._routes.advance_tick(tick), 1, "headless source/root tick")
		assert_equal(driver.read_route_into(actual._routes, 5, job, frame), &"", "actual source consumer")
		assert_equal(actual._routes.source_state_into(actual._worker, job, 5, 1, 2, state), &"", "canonical state")
		var original: PackedInt64Array = state.duplicate()
		assert_equal(driver.read_route_into(actual._routes, 5, job, frame), &"", "additional render read")
		assert_equal(actual._routes.source_state_into(actual._worker, job, 5, 1, 2, state), &"", "state after render")
		assert_equal(state, original, "rendering never advances canonical phase")
		if state[0] == Driver.SourceProgram.WALK:
			assert_equal(source.clip_into(1, state[1], expected), &"", "exact Content resolver")
			assert_equal(frame.frames.slice(0, 3), expected, "forward source phase")
	assert_true(frame.ready, "actual canonical ready terminal")
	assert_equal(driver.step_profile_into(5, job, Driver.ONE, false, frame), &"MOLE_DRIVER_CANONICAL_ROUTES_REQUIRED", "legacy render clock cannot reopen programme5")
	assert_equal(actual._routes.refresh_work_actor(actual._worker, job, 24, 1, 2, 0, -1, tool), &"", "source-ready exact FRONT entry")
	for tick: int in range(47, 113):
		assert_equal(actual._routes.advance_tick(tick), 1, "headless entry followed by one full work loop")
		assert_equal(driver.read_route_into(actual._routes, 24, job, frame), &"", "real work source consumer")
		assert_equal(actual._routes.source_state_into(actual._worker, job, 24, 1, 2, state), &"", "canonical work source state")
		assert_equal(source.clip_into(9 if state[0] == Driver.SourceProgram.ENTRY else 8, state[1], expected), &"", "exact entry/work Content phase")
		assert_equal(frame.frames.slice(0, 3), expected, "actual source palette phase")
	assert_equal(actual._routes.request_source_ready(actual._worker, job), &"", "canonical productive stop")
	for tick: int in range(113, 143):
		assert_equal(actual._routes.advance_tick(tick), 1, "headless full recovery before retreat")
		assert_equal(driver.read_route_into(actual._routes, 24, job, frame), &"", "actual recovery source consumer")
	assert_true(frame.ready, "productive source recovered without rendering authority")
	assert_equal(actual._routes.refresh_travel_actor(actual._worker, job, 9, 1, 2, 0, -1, tool), &"", "ready-only backward handoff")
	assert_equal(actual._routes.request_route(actual._worker, first, 143), &"", "actual retreat edge")
	for tick: int in range(143, 189):
		assert_equal(actual._routes.advance_tick(tick), 1, "headless reverse source/root tick")
		assert_equal(driver.read_route_into(actual._routes, 9, job, frame), &"", "backward actual source")
		assert_equal(actual._routes.source_state_into(actual._worker, job, 9, 1, 2, state), &"", "original source state")
		if state[0] == Driver.SourceProgram.WALK:
			assert_equal(source.clip_into(1, (Driver.SourceProgram.WALK_DURATION - state[1]) % Driver.SourceProgram.WALK_DURATION, expected), &"", "reverse the exact source time")
			assert_equal(frame.frames.slice(0, 3), expected, "backward source includes short final interval")
		assert_equal(frame.yaw, 49152, "backward root does not turn body")
	assert_equal(frame.point, a, "real backward root returns to the access")
	assert_true(frame.ready, "source ends at shared ready")

func _content(compact: bool = false, installation: bool = false) -> Content:
	"""Versioned finite clips use one actual ArrayMesh; source flags remain a labelled synthetic fixture."""
	var helper: MeshFixture = MeshFixture.new()
	var mesh: ArrayMesh = helper._mesh()
	var counts: Array[int] = COUNTS.duplicate()
	if compact:
		counts.append_array([4, 5, 5])
	if installation:
		assert_true(compact, "INSTALL extends the compact source, never the original program")
		counts.append_array([4, 5, 5])
	var frames: int = 0
	for count: int in counts:
		frames += count
	var path: String = "user://mole-driver-%d.bin" % get_instance_id()
	var file: FileAccess = FileAccess.open(path, FileAccess.WRITE)
	file.store_buffer("UGACNT01".to_ascii_buffer())
	for value: int in [1, 1, 1, counts.size(), frames, 12]:
		file.store_32(value)
	for value: int in [0, -32, 0, 4096, 4096, 4096]:
		file.store_32(value)
	for which: int in 4:
		file.store_buffer(MeshFixture.HASH.hex_decode())
	helper._write_part(file, mesh, 0)
	_write_clips(file, counts)
	for frame: int in frames:
		for value: float in PackedFloat32Array([1, 0, 0, 0, 1, 0, 0, 0, 1, 0, 0, 0]):
			file.store_float(value)
	for frame: int in frames:
		file.store_float(0.0)
	file.store_buffer("UGAEND01".to_ascii_buffer())
	file.close()
	var result: Content = Content.new()
	assert_equal(result.load_file(path, FileAccess.get_sha256(path), 3 * 1024 * 1024), &"", "real immutable source loader")
	assert_equal(DirAccess.remove_absolute(ProjectSettings.globalize_path(path)), OK, "owned fixture removed")
	return result


func _write_clips(file: FileAccess, counts: Array[int]) -> void:
	"""Finite profile protocol order with actual loop rules and exact matching source endpoints."""
	var first: int = 0
	for clip: int in counts.size():
		var loop: int = int(clip < 2 or (clip - 2) % 3 == 0)
		for value: int in [first, counts[clip], loop, (counts[clip] - 1) * Driver.ONE]:
			file.store_32(value)
		file.store_buffer(MeshFixture.HASH.hex_decode())
		first += counts[clip]


func _install_profiles(revision: int = 1, count: int = 4, bad_heading_row: int = -1) -> void:
	"""Actual compiled mole/tool keys plus test-only geometry and certificate bits."""
	var rows: Array[Dictionary] = []
	var boxes: Array[PackedInt32Array] = []
	for index: int in count:
		var row: Dictionary = _fixture._row(index if index < 2 else Profiles.MODE_WORK)
		row.fields[Profiles.F_TOOL] = _fixture._items.compiled_id(&"tool")
		row.fields[Profiles.F_TOOL_VARIANT] = _fixture.Gear.MANUFACTURE_BASIC
		row.fields[Profiles.F_FIRST_BOX] = boxes.size()
		row.fields[Profiles.F_BOX_COUNT] = 3 if index < 2 else 7
		boxes.append_array(_fixture._boxes())
		if index >= 2:
			row.fields[Profiles.F_YAW_KIND] = Profiles.YAW_EXACT
			if count == 18:
				@warning_ignore("integer_division") var heading: int = (index - 2) / 4
				row.fields[Profiles.F_YAW] = heading * 16384 + int(index == bad_heading_row)
			row.fields[Profiles.F_WORK_KIND] = _fixture.Jobs.JOB_KIND_BUILD
			row.fields[Profiles.F_CONTACT_KIND] = Profiles.CONTACT_ANCHOR_AND_PATCH
			_add_work_boxes(boxes, 2 + (index - 2) % 4 if count == 18 else index)
		rows.append(row)
	var wire: PackedByteArray = _fixture._image(rows, boxes, revision)
	var digest: PackedByteArray = _source.source_digest().hex_decode()
	for byte: int in 32:
		wire[32 + byte] = digest[byte]
	assert_equal(_fixture._load(wire, revision), &"", "complete synthetic rows through actual immutable reader")


func _add_work_boxes(boxes: Array[PackedInt32Array], index: int) -> void:
	"""Two visibly different face axes exist for one actual worker and one tool manufacture."""
	boxes.append(PackedInt32Array([-128, -20, -128, 128, 1000, 256, Profiles.WORK_APPROACH]))
	boxes.append(PackedInt32Array([-128, 0, -256, 128, 1000, 128, Profiles.WORK_STROKE]))
	boxes.append(PackedInt32Array([0, 512, -200, 0, 512, -200, Profiles.CONTACT_POINT]))
	boxes.append(PackedInt32Array([-2, 512, -202, 2, 512, -198, Profiles.CONTACT_PATCH]) if index == 2 else
		PackedInt32Array([-2, 510, -200, 2, 514, -200, Profiles.CONTACT_PATCH]))
