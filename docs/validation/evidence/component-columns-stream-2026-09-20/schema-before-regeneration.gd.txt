## Section 4 COMPONENT_COLUMNS immutable metadata (SAVE-S4-STREAM-R01 v2, ADR 0169).
##
## The compiled metadata for section 4's first implemented body: 18 owner blocks in ASCII key
## order, 298 owner-local fields, five independent child extents, 12947565 bytes over 193184
## descriptor rows. tools/generate_component_columns_schema.py compiles the marked region below
## from docs/planning/component_columns_layout.json plus the canonical registry, after re-proving
## all 298 counts against the section-4 source-capacity equalities. Nothing here reads JSON at
## runtime, instantiates a live store, or calls an owner module.
##
## Every index is section-local and `field` is the ordinal WITHIN its owner, never a position in a
## type bucket; storage_index() reports that separately. Invalid lookups return an empty String or
## zero, and ZERO IS NOT AN ERROR SENTINEL -- no real count, extent or offset here is zero. Every
## stream and accessor caller MUST call owner_valid(), field_valid() or child_valid() first.
##
## No framing, cursor, record type, byte-order probe or semantic validator lives here; those belong
## to save_section_component_columns.gd and to the 18 owner adapters SAVE-S4-CODEC still requires.
##
## The tables are an added IMMUTABLE METADATA LOGICAL PAYLOAD -- 18 owner rows across 11 columns,
## 298 field rows across 3 columns and 5 child extents -- charged SEPARATELY from the 6417408-byte
## owner plus transient allowance, and not a measured resident-set figure.

const SaveHeader := preload("res://scripts/core/save_header.gd")

## Pinned by hand, outside the generated region, so a regenerated table that moved the declaration,
## the section schema or a total is refused rather than believed.
const EXPECTED_DECLARATION_VERSION: int = 1
const EXPECTED_SECTION_SCHEMA_VERSION: int = 2
const EXPECTED_STORE_COUNT: int = 18
const EXPECTED_FIELD_COUNT: int = 298
const EXPECTED_SECTION_BYTES: int = 12947565
const EXPECTED_DESCRIPTOR_ROW_COUNT: int = 193184

## The bounded fragment window both cursors use. Not a layout value.
const CHUNK_BYTES: int = 65536

## Framing widths: store_count u32; wrapper key-length u32 + version u32 + primary u64 +
## payload_length u64 = 24 fixed bytes beside the key text; child_extent_count u32; each child
## extent u64; each field element_count u64.
const STORE_COUNT_BYTES: int = 4
const OWNER_HEADER_FIXED_BYTES: int = 24
const CHILD_COUNT_BYTES: int = 4
const CHILD_EXTENT_BYTES: int = 8
const FIELD_COUNT_BYTES: int = 8

## The only value types section 4 declares. There is no u32, u64 or string value field.
const TYPE_U8: int = 0
const TYPE_I32: int = 2
const TYPE_I64: int = 4

## Only the codes this module itself returns; the streaming module owns the other SAVE_COMPONENT_*.
const REFUSE_NONE: StringName = &""
const REFUSE_SCHEMA: StringName = &"SAVE_COMPONENT_SCHEMA"
const REFUSE_METADATA: StringName = &"SAVE_COMPONENT_METADATA"

# --- validity predicates ---------------------------------------------------------------------


static func owner_valid(owner: int) -> bool:
	"""True when `owner` is one of the 18 section-local owner indexes."""
	return owner >= 0 and owner < STORE_COUNT


static func field_valid(owner: int, field: int) -> bool:
	"""True when `field` is an owner-local ordinal of a valid owner."""
	return owner_valid(owner) and field >= 0 and field < int(OWNER_FIELD_COUNTS[owner])


static func child_valid(owner: int, child: int) -> bool:
	"""True when `child` indexes one of this owner's declared independent child extents."""
	return owner_valid(owner) and child >= 0 and child < int(OWNER_CHILD_COUNTS[owner])


# --- owner metadata --------------------------------------------------------------------------


static func owner_key(owner: int) -> String:
	"""ASCII owner key, or an empty String on an invalid index."""
	if not owner_valid(owner):
		return ""
	return String(OWNER_KEYS[owner])


static func owner_version(owner: int) -> int:
	"""This owner's section 4 schema version, which is its own and not another section's."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_VERSIONS[owner])


static func primary_count(owner: int) -> int:
	"""Exact declared primary row count. Not an entity count, and never inferred from a field."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_PRIMARY_COUNTS[owner])


static func child_extent_count(owner: int) -> int:
	"""Number of INDEPENDENT child extents declared; fixed derived strides are not extents."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_CHILD_COUNTS[owner])


static func child_extent(owner: int, child: int) -> int:
	"""One declared child extent, exact and never compacted."""
	if not child_valid(owner, child):
		return 0
	return int(CHILD_EXTENTS[int(OWNER_CHILD_BEGIN[owner]) + child])


static func field_count(owner: int) -> int:
	"""Number of fields this owner declares, in owner-local ordinal order."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_FIELD_COUNTS[owner])


# --- field metadata --------------------------------------------------------------------------


static func field_key(owner: int, field: int) -> String:
	"""ASCII field key at this owner-local ordinal, or an empty String when invalid."""
	if not field_valid(owner, field):
		return ""
	return String(FIELD_KEYS[int(OWNER_FIELD_BEGIN[owner]) + field])


static func field_type(owner: int, field: int) -> int:
	"""Type code 0 (u8), 2 (i32) or 4 (i64). Invalid gives -1, since 0 is a real type code."""
	if not field_valid(owner, field):
		return -1
	return int(FIELD_TYPES[int(OWNER_FIELD_BEGIN[owner]) + field])


static func storage_index(owner: int, field: int) -> int:
	"""Count of EARLIER fields of the SAME type in this owner: the column's bucket position."""
	if not field_valid(owner, field):
		return 0
	var begin: int = int(OWNER_FIELD_BEGIN[owner])
	var code: int = int(FIELD_TYPES[begin + field])
	var seen: int = 0
	for earlier: int in field:
		if int(FIELD_TYPES[begin + earlier]) == code:
			seen += 1
	return seen


static func element_count(owner: int, field: int) -> int:
	"""Exact element count of this column; every column covers its full physical capacity."""
	if not field_valid(owner, field):
		return 0
	return int(FIELD_COUNTS[int(OWNER_FIELD_BEGIN[owner]) + field])


static func field_width(owner: int, field: int) -> int:
	"""Bytes per element of this field: 1, 4 or 8."""
	if not field_valid(owner, field):
		return 0
	return _type_width(int(FIELD_TYPES[int(OWNER_FIELD_BEGIN[owner]) + field]))


static func _type_width(type_code: int) -> int:
	"""Width of a supported type code, or 0 for a code section 4 does not declare."""
	if type_code == TYPE_U8:
		return 1
	if type_code == TYPE_I32:
		return 4
	if type_code == TYPE_I64:
		return 8
	return 0


# --- byte geometry, relative to section start --------------------------------------------------


static func payload_bytes(owner: int) -> int:
	"""Declared payload length: child header, child extents, field counts and all values."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_PAYLOAD_BYTES[owner])


static func owner_block_bytes(owner: int) -> int:
	"""Whole owner block: wrapper words, key text and payload."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_BLOCK_BYTES[owner])


static func owner_offset(owner: int) -> int:
	"""Offset of this owner's block; the first owner follows the store_count fragment."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_OFFSETS[owner])


static func _key_bytes(owner: int) -> int:
	"""UTF-8 byte length of the owner key as framed after its u32 length prefix."""
	if not owner_valid(owner):
		return 0
	return int(OWNER_KEY_BYTES[owner])


static func _payload_offset(owner: int) -> int:
	"""Offset of this owner's payload, immediately after its wrapper words and key text."""
	return owner_offset(owner) + OWNER_HEADER_FIXED_BYTES + _key_bytes(owner)


static func field_count_offset(owner: int, field: int) -> int:
	"""Offset of this field's 8-byte element_count fragment."""
	if not field_valid(owner, field):
		return 0
	var offset: int = _payload_offset(owner) + CHILD_COUNT_BYTES \
		+ CHILD_EXTENT_BYTES * child_extent_count(owner)
	var begin: int = int(OWNER_FIELD_BEGIN[owner])
	for earlier: int in field:
		var index: int = begin + earlier
		offset += FIELD_COUNT_BYTES + int(FIELD_COUNTS[index]) * _type_width(int(FIELD_TYPES[index]))
	return offset


static func field_value_offset(owner: int, field: int) -> int:
	"""Offset of this field's first value byte, immediately after its element_count."""
	if not field_valid(owner, field):
		return 0
	return field_count_offset(owner, field) + FIELD_COUNT_BYTES


# --- compiled metadata validation --------------------------------------------------------------


static func schema_refusal() -> SaveHeader.Refusal:
	"""Refuse unless the declaration, the section schema, the pinned totals, the table shape and
	every compiled arithmetic relation agree, and every type is u8, i32 or i64."""
	if DECLARATION_VERSION != EXPECTED_DECLARATION_VERSION \
			or SECTION_SCHEMA_VERSION != EXPECTED_SECTION_SCHEMA_VERSION:
		return _refuse(REFUSE_SCHEMA, "declaration %d / section schema %d is not %d / %d"
			% [DECLARATION_VERSION, SECTION_SCHEMA_VERSION, EXPECTED_DECLARATION_VERSION,
				EXPECTED_SECTION_SCHEMA_VERSION])
	if STORE_COUNT != EXPECTED_STORE_COUNT or FIELD_COUNT != EXPECTED_FIELD_COUNT \
			or SECTION_BYTES != EXPECTED_SECTION_BYTES \
			or DESCRIPTOR_ROW_COUNT != EXPECTED_DESCRIPTOR_ROW_COUNT:
		return _refuse(REFUSE_SCHEMA,
			"compiled totals %d owners / %d fields / %d bytes / %d rows are not the pinned body"
				% [STORE_COUNT, FIELD_COUNT, SECTION_BYTES, DESCRIPTOR_ROW_COUNT])
	var shape: SaveHeader.Refusal = _table_shape_refusal()
	if not shape.is_ok():
		return shape
	return _table_arithmetic_refusal()


static func _table_shape_refusal() -> SaveHeader.Refusal:
	"""Every owner column holds 18 rows and every field column holds 298 rows."""
	var owner_columns: Array = [OWNER_KEYS, OWNER_KEY_BYTES, OWNER_VERSIONS, OWNER_PRIMARY_COUNTS,
		OWNER_FIELD_COUNTS, OWNER_FIELD_BEGIN, OWNER_CHILD_COUNTS, OWNER_CHILD_BEGIN,
		OWNER_OFFSETS, OWNER_PAYLOAD_BYTES, OWNER_BLOCK_BYTES]
	for column: Array in owner_columns:
		if column.size() != STORE_COUNT:
			return _refuse(REFUSE_METADATA, "an owner column holds %d of %d rows"
				% [column.size(), STORE_COUNT])
	for column: Array in [FIELD_KEYS, FIELD_TYPES, FIELD_COUNTS]:
		if column.size() != FIELD_COUNT:
			return _refuse(REFUSE_METADATA, "a field column holds %d of %d rows"
				% [column.size(), FIELD_COUNT])
	return _accept()


static func _table_arithmetic_refusal() -> SaveHeader.Refusal:
	"""Walk the 18 owners once, rebuilding spans, offsets and totals from the framing rules."""
	var fields: int = 0
	var children: int = 0
	var offset: int = STORE_COUNT_BYTES
	var rows: int = 0
	for owner: int in STORE_COUNT:
		var owner_refusal: SaveHeader.Refusal = _owner_refusal(owner, fields, children, offset)
		if not owner_refusal.is_ok():
			return owner_refusal
		fields += field_count(owner)
		children += child_extent_count(owner)
		offset += owner_block_bytes(owner)
		rows += primary_count(owner)
	if fields != FIELD_COUNT or children != CHILD_EXTENTS.size():
		return _refuse(REFUSE_METADATA, "owner spans cover %d fields and %d child extents"
			% [fields, children])
	if offset != SECTION_BYTES or rows != DESCRIPTOR_ROW_COUNT:
		return _refuse(REFUSE_METADATA, "compiled blocks total %d bytes over %d rows"
			% [offset, rows])
	return _accept()


static func _owner_refusal(owner: int, fields: int, children: int,
		offset: int) -> SaveHeader.Refusal:
	"""One owner's span cursors, identity and payload arithmetic."""
	if int(OWNER_FIELD_BEGIN[owner]) != fields or int(OWNER_CHILD_BEGIN[owner]) != children \
			or owner_offset(owner) != offset:
		return _refuse(REFUSE_METADATA,
			"owner %d declares span %d/%d at %d; the walk reached %d/%d at %d"
				% [owner, int(OWNER_FIELD_BEGIN[owner]), int(OWNER_CHILD_BEGIN[owner]),
					owner_offset(owner), fields, children, offset])
	var identity: SaveHeader.Refusal = _owner_identity_refusal(owner)
	if not identity.is_ok():
		return identity
	return _owner_payload_refusal(owner)


static func _owner_identity_refusal(owner: int) -> SaveHeader.Refusal:
	"""Key text, key byte length, owner version, primary count and strict ASCII owner order."""
	var key: String = String(OWNER_KEYS[owner])
	if key.is_empty() or key.to_utf8_buffer().size() != _key_bytes(owner):
		return _refuse(REFUSE_METADATA, "owner %d key '%s' is not %d UTF-8 bytes"
			% [owner, key, _key_bytes(owner)])
	if owner_version(owner) < 1 or primary_count(owner) <= 0:
		return _refuse(REFUSE_METADATA, "owner '%s' declares version %d and %d primaries"
			% [key, owner_version(owner), primary_count(owner)])
	if owner > 0 and String(OWNER_KEYS[owner - 1]) >= key:
		return _refuse(REFUSE_METADATA, "owner '%s' does not follow '%s' in ASCII order"
			% [key, String(OWNER_KEYS[owner - 1])])
	return _accept()


static func _owner_payload_refusal(owner: int) -> SaveHeader.Refusal:
	"""Rebuild this owner's payload and block length from its extents, types and counts."""
	var begin: int = int(OWNER_FIELD_BEGIN[owner])
	var wrapper: int = OWNER_HEADER_FIXED_BYTES + _key_bytes(owner) + CHILD_COUNT_BYTES \
		+ CHILD_EXTENT_BYTES * child_extent_count(owner)
	var bytes: int = CHILD_COUNT_BYTES + CHILD_EXTENT_BYTES * child_extent_count(owner)
	for child: int in child_extent_count(owner):
		if child_extent(owner, child) <= 0:
			return _refuse(REFUSE_METADATA, "owner '%s' child extent %d is not positive"
				% [owner_key(owner), child])
	for ordinal: int in field_count(owner):
		var index: int = begin + ordinal
		var width: int = _type_width(int(FIELD_TYPES[index]))
		if width == 0:
			return _refuse(REFUSE_SCHEMA, "field '%s' declares unsupported type %d"
				% [String(FIELD_KEYS[index]), int(FIELD_TYPES[index])])
		if int(FIELD_COUNTS[index]) <= 0 or String(FIELD_KEYS[index]).is_empty():
			return _refuse(REFUSE_METADATA, "field %d of owner '%s' declares count %d"
				% [ordinal, owner_key(owner), int(FIELD_COUNTS[index])])
		bytes += FIELD_COUNT_BYTES + int(FIELD_COUNTS[index]) * width
	if bytes != payload_bytes(owner) \
			or owner_block_bytes(owner) != OWNER_HEADER_FIXED_BYTES + _key_bytes(owner) + bytes:
		return _refuse(REFUSE_METADATA, "owner '%s' declares %d/%d bytes; framing gives %d/%d"
			% [owner_key(owner), payload_bytes(owner), owner_block_bytes(owner), bytes,
				OWNER_HEADER_FIXED_BYTES + _key_bytes(owner) + bytes])
	if wrapper > CHUNK_BYTES:
		return _refuse(REFUSE_METADATA, "owner '%s' wrapper is %d bytes, over the %d window"
			% [owner_key(owner), wrapper, CHUNK_BYTES])
	return _accept()


static func _refuse(code: StringName, detail: String) -> SaveHeader.Refusal:
	"""Build a refusal carrying this module's own code and detail."""
	return SaveHeader.Refusal.new(code, detail)


static func _accept() -> SaveHeader.Refusal:
	"""The accepted result: an empty code and no detail."""
	return SaveHeader.Refusal.new(REFUSE_NONE, "")
