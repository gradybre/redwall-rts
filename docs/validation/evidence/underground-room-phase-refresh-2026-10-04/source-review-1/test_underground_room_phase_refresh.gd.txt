class Binding extends Phase.PhaseBinding:
	var ordinary: Vector2i = NULL_REF
	var prepare_fault: int = 0
	var saw_null_placement: bool = false

	func prepare_companions(token: int, site: Vector2i, op: int, stage: int,
			room: Vector2i, plan: Space.Plan) -> int:
		"""Entry keeps its existing API; only the ordinary actual Room invokes the distinct new path."""
		if room != ordinary and prepare_fault != 10: return super.prepare_companions(token, site, op, stage, room, plan)
		var actual: ActualAuthority = spatial.get_ref() as ActualAuthority
		if prepare_fault == 1: site.y += 1
		if prepare_fault == 2: room.y += 1
		if prepare_fault == 3: token += 1
		if prepare_fault == 4: stage = Contract.STAGE_CANCEL
		if prepare_fault == 5: op = Contract.OP_CUT
		if prepare_fault == 7: actual = ActualAuthority.new()
		if prepare_fault == 8: token = 0
		if prepare_fault == 9: site = NULL_REF
		var result: int = placements.prepare_room_phase_refresh(actual, site, op, stage, room, token,
			cold_active + (1 if prepare_fault == 6 else 0))
		saw_null_placement = result > 0 and placements._phase_context.placement == NULL_REF
		return result

	func prepared_refusal(token: int) -> StringName:
		"""Run the full actual source/geometry/retention observations under the original ordinary tuple."""
		if placements._phase_context.placement != NULL_REF: return super.prepared_refusal(token)
		return (spatial.get_ref() as ActualAuthority).room_phase_observation_refusal(token)

	func revision_after(token: int) -> int:
		"""Synthetic qualification is explicit and independent of whether a connector Frontier exists."""
		return qualified_revision if placements.phase_context_leaf_refusal(token) == &"" else 0

	func phase_final_leaf_refusal(site: Vector2i, op: int, stage: int,
			token: int, space: int, companion: int) -> StringName:
		"""Every original scalar remains exact; the adverse callback models a provider lying after its own leaf."""
		if placements._phase_context.placement != NULL_REF:
			return super.phase_final_leaf_refusal(site, op, stage, token, space, companion)
		var actual: ActualAuthority = spatial.get_ref() as ActualAuthority
		var context: Locations.PhaseContext = actual.room_phase_context(companion)
		if context == null or context.site != site or context.operation != op or context.stage != stage \
				or context.cold_token != token or context.space_token != space: return &"TEST_ROOM_PHASE_CONTEXT"
		var code: StringName = ActualAuthority.room_phase_leaf_refusal(actual, companion)
		if leaf_probe.is_valid():
			var probe: Callable = leaf_probe
			leaf_probe = Callable()
			probe.call()
		return code

	func discard_companions(token: int) -> void:
		"""Discard exact original candidates through the matching public ordinary or entry path."""
		if placements._prepared_placement != NULL_REF: super.discard_companions(token)
		else: (spatial.get_ref() as ActualAuthority).discard_room_phase_refresh(token)

	_setup_room(Buildings.ROOM_TYPE_CORRIDOR)
	var ordinary_site: Vector2i = _economic._site
	assert_equal(_placements._ordinary_room_refusal(_f.other_corridor), &"", "ordinary Tunnel is allowed")
	assert_true(_placements._ordinary_room_refusal(_f.corridor) != &"", "real Entry has its own path")
	assert_true(_economic._complete(Contract.OP_BRACE) > 0, "real ordinary Corridor paid phase")
	assert_true(_economic.failures.is_empty(), "ordinary Corridor: %s" % _economic.failures)
	_select_economic_room(_entry_site, _f.corridor, _f.corridor_floor)
	_ready_start()
	_phase.prepare_fault = 10
	var before: Dictionary = _state()
	assert_false(_sites.begin_phase_work(_entry_site, 0).ok, "Entry cannot borrow the ordinary null branch")
	_unchanged(before)
	_phase.prepare_fault = 0
	assert_true(_sites.begin_phase_work(_entry_site, 0).ok, "real Entry uses its existing non-null path")
	assert_true(_sites.cancel_phase(_entry_site, _economic._store).ok, "same exact funded Entry cleanup")
	_select_economic_room(ordinary_site, _f.other_corridor, _f.other_floor)


func test_changed_borrowed_context_fields_refuse_and_original_cleanup_still_releases() -> void:
	"""A mutable borrowed packet cannot change branch selection or strand its independently pinned original tokens."""
	_ready_start()
	var before: Dictionary = _state()
	for field: StringName in [&"placement", &"site", &"room", &"space_token", &"location_token", &"route_token"]:
		_phase.final_probe = _change_context_field.bind(field)
		assert_false(_sites.begin_phase_work(_economic._site, 0).ok, "changed borrowed %s" % field)
		_unchanged(before)
		assert_true(_f._budget.is_quiescent(), "original cleanup survives changed packet %s" % field)
		assert_equal(_f._locations._token, 0, "exact Location candidate dropped")
		assert_equal(_f._routes._token, 0, "exact Route candidate dropped")
	assert_true(_sites.begin_phase_work(_economic._site, 0).ok, "unchanged original operation retries")


func _change_context_field(field: StringName) -> void:
	"""Only the public borrowed operation fields are altered; private original issuer controls stay intact."""
	var context: Locations.PhaseContext = _placements._phase_context
	if field == &"placement": context.placement = _placement
	elif field == &"site": context.site.y += 1
	elif field == &"room": context.room.y += 1
	elif field == &"space_token": context.space_token += 1
	elif field == &"location_token": context.location_token += 1
	elif field == &"route_token": context.route_token += 1
